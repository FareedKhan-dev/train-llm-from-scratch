# Post-training stack for the from-scratch SFT / RM / PPO / DPO / GRPO suite. Install into
# a fresh venv:
#
#   python3 -m venv .venv && source .venv/bin/activate
#   pip install -r requirements-post.txt
#
# `pip install -e ".[train]"` installs the same packages. torchrun (multi-GPU) ships with
# torch; see requirements.txt for picking a CUDA build of torch.

torch
numpy
h5py
tqdm
tiktoken
zstandard
jaxtyping         # tensor shape annotations
regex             # the BPE tokenizer's split pattern
datasets          # HuggingFace dataset downloads (Alpaca, HH-RLHF, GSM8K, ...)
wandb             # optional experiment logging (every script also writes JSONL)
